package main
import("context";"fmt";"runtime";t "github.com/mattjoyce/tarski")
func main(){for _,n:=range []int{1000,100000}{s:=t.Schema{Arities:map[string]int{"edge":1}}; fs:=make([]t.Fact,n);for i:=range fs {fs[i]=t.Fact{Relation:"edge",Tuple:t.Tuple{t.Int(int64(i))}}};b,e:=t.NewBase(s,fs...);if e!=nil{panic(e)};pr,e:=t.Parse("out(X) :- edge(X).");if e!=nil{panic(e)};p,e:=t.Compile(s,pr);if e!=nil{panic(e)};runtime.GC();var before,after runtime.MemStats;runtime.ReadMemStats(&before);_,e=t.Evaluate(context.Background(),p,b,t.Options{MaxSteps:1,MaxBytes:1});runtime.ReadMemStats(&after);fmt.Printf("n=%d evaluation allocation=%d err=%v\n",n,after.TotalAlloc-before.TotalAlloc,e)}}
